void main() {
  // Variables and Data Types
  String name = "Dwij";
  int age = 20;
  double cgpa = 8.75;
  bool isStudent = true;
  
  print("Name: $name");
  print("Age: $age");
  print("CGPA: $cgpa");
  print("Student: $isStudent");

  // Control Flow
  if (age >= 18) {
    print("\nEligible to vote.");
  } else {
    print("\nNot eligible to vote.");
  }

  print("\nFor Loop:");
  for (int i = 1; i <= 5; i++) {
    print("Count: $i");
  }

  // Functions
  int sum(int a, int b) {
    return a + b;
  }
  print("\nSum of 10 and 20 = ${sum(10, 20)}");

  // List
  List<String> fruits = ["Apple", "Banana", "Mango"];
  print("\nList:");
  for (var fruit in fruits) {
    print(fruit);
  }

  // Map
  Map<String, int> marks = {
    "Math": 90,
    "Science": 88,
    "English": 92
  };
  print("\nMap:");
  marks.forEach((subject, mark) {
    print("$subject: $mark");
  });

  // Set
  Set<int> numbers = {10, 20, 30, 20, 10};
  print("\nSet:");
  print(numbers);

  // Null Safety
  String? city;
  print("\nNullable String:");
  print(city);
  city = "Ahmedabad";
  print(city);

  String country = "India";
  print(country);
}
