import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: RegistrationScreen(),
    );
  }
}

class RegistrationScreen extends StatefulWidget {
  const RegistrationScreen({super.key});

  @override
  State<RegistrationScreen> createState() => RegistrationScreenState();
}

class RegistrationScreenState extends State<RegistrationScreen> {
  String name = "";
  String username = "";
  String email = "";
  String password = "";
  String errorMessage = "";

  void validateAndSubmit() {
    setState(() {
      if (name.isEmpty || username.isEmpty || email.isEmpty || password.isEmpty) {
        errorMessage = 'Error: All fields are required!';
      } else if (username.length < 3) {
        errorMessage = 'Error: Username must be at least 3 letters!';
      } else {
        errorMessage = "";
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (context) => SuccessScreen(newUserName: username),
          ),
        );
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFF940D0D),
      appBar: AppBar(
        title: const Text(
          'Register',
          style: TextStyle(
            color: Color(0xFFE09E02),
            fontWeight: FontWeight.bold,
          ),
        ),
        backgroundColor: const Color(0xFF940D0D),
        iconTheme: const IconThemeData(color: Color(0xFFE09E02)),
      ),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text(
              'Enter Details',
              style: TextStyle(fontSize: 24, color: Color(0xFFE09E02)),
            ),
            Container(height: 20),
            TextField(
              style: const TextStyle(color: Color(0xFFE09E02)),
              decoration: const InputDecoration(
                labelText: 'Name',
                labelStyle: TextStyle(color: Color(0xFFE09E02)),
                enabledBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
                focusedBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
              ),
              onChanged: (value) {
                name = value;
              },
            ),
            Container(height: 10),
            TextField(
              style: const TextStyle(color: Color(0xFFE09E02)),
              decoration: const InputDecoration(
                labelText: 'Username',
                labelStyle: TextStyle(color: Color(0xFFE09E02)),
                enabledBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
                focusedBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
              ),
              onChanged: (value) {
                username = value;
              },
            ),
            Container(height: 10),
            TextField(
              style: const TextStyle(color: Color(0xFFE09E02)),
              decoration: const InputDecoration(
                labelText: 'Email',
                labelStyle: TextStyle(color: Color(0xFFE09E02)),
                enabledBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
                focusedBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
              ),
              onChanged: (value) {
                email = value;
              },
            ),
            Container(height: 10),
            TextField(
              obscureText: true,
              style: const TextStyle(color: Color(0xFFE09E02)),
              decoration: const InputDecoration(
                labelText: 'Password',
                labelStyle: TextStyle(color: Color(0xFFE09E02)),
                enabledBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
                focusedBorder: UnderlineInputBorder(
                  borderSide: BorderSide(color: Color(0xFFE09E02)),
                ),
              ),
              onChanged: (value) {
                password = value;
              },
            ),
            Container(height: 20),
            Text(
              errorMessage,
              style: const TextStyle(color: Color(0xFFE09E02), fontSize: 16),
            ),
            Container(height: 20),
            ElevatedButton(
              style: ElevatedButton.styleFrom(
                backgroundColor: const Color(0xFF940D0D),
                foregroundColor: const Color(0xFFE09E02),
                side: const BorderSide(color: Color(0xFFE09E02)),
              ),
              onPressed: validateAndSubmit,
              child: const Text('SUBMIT'),
            ),
          ],
        ),
      ),
    );
  }
}

class SuccessScreen extends StatelessWidget {
  final String newUserName;
  SuccessScreen({super.key, required this.newUserName});
  final List<String> userList = ['Alice', 'Bob', 'Charlie'];

  @override
  Widget build(BuildContext context) {
    List<String> displayList = List.from(userList);
    displayList.add(newUserName);

    return Scaffold(
      backgroundColor: const Color(0xFF940D0D),
      appBar: AppBar(
        title: const Text('Success', style: TextStyle(color: Color(0xFFE09E02))),
        backgroundColor: const Color(0xFF940D0D),
        iconTheme: const IconThemeData(color: Color(0xFFE09E02)),
      ),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              'Welcome, $newUserName!',
              style: const TextStyle(fontSize: 24, color: Color(0xFFE09E02)),
            ),
            Container(height: 20),
            const Text(
              'All Registered Users:',
              style: TextStyle(fontSize: 18, color: Color(0xFFE09E02)),
            ),
            Container(height: 10),
            for (String user in displayList)
              Container(
                margin: const EdgeInsets.only(bottom: 10),
                padding: const EdgeInsets.all(10),
                decoration: BoxDecoration(
                  border: Border.all(color: const Color(0xFFE09E02)),
                ),
                child: Row(
                  children: [
                    const Icon(Icons.person, color: Color(0xFFE09E02)),
                    Container(width: 10),
                    Text(
                      user,
                      style: const TextStyle(fontSize: 16, color: Color(0xFFE09E02)),
                    )
                  ],
                ),
              ),
            Container(height: 30),
            ElevatedButton(
              style: ElevatedButton.styleFrom(
                backgroundColor: const Color(0xFF940D0D),
                foregroundColor: const Color(0xFFE09E02),
                side: const BorderSide(color: Color(0xFFE09E02)),
              ),
              onPressed: () {
                Navigator.pop(context);
              },
              child: const Text('GO BACK'),
            ),
          ],
        ),
      ),
    );
  }
}
